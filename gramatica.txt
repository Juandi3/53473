GRAMATICA ASIGNADA - TEMA 39568_10
Notacion ANSI-C (BNF)
======================================

programa: instrucciones;

instrucciones:
    instruccion
    | instrucciones instruccion
    ;

instruccion: bucle;

bucle:
    "do" "{" sentencia "}" "while" "(" condicion ")" ";"
    ;

sentencia:
    salida
    | salida sentencia
    | terminar
    ;

salida:
    "puts" "(" cadena ")" ";"
    ;

terminar:
    "break" ";"
    ;

condicion:
    "0"
    | "1"
    ;

cadena:
    "\"" caracteres "\""
    ;

caracteres:
    caracter
    | caracteres caracter
    ;

caracter:
    letra
    | digito
    | simbolo
    ;

letra:
    "a" ... "z"
    | "A" ... "Z"
    ;

digito:
    "0" ... "9"
    ;

simbolo:
    "." | "#" | "!" | "?" | ":" | ";" | " "
    ;
